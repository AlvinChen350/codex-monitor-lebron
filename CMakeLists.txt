cmake_minimum_required(VERSION 3.24)
project(codex_monitor VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(vterm
    GIT_REPOSITORY https://github.com/neovim/libvterm.git
    GIT_TAG 934bc2fbf21800ac3458a499df8820ca5fb45fd3)
FetchContent_MakeAvailable(vterm)
file(GLOB VTERM_SOURCES CONFIGURE_DEPENDS "${vterm_SOURCE_DIR}/src/*.c")
add_library(vterm STATIC ${VTERM_SOURCES})
target_include_directories(vterm PUBLIC "${vterm_SOURCE_DIR}/include")
target_compile_definitions(vterm PRIVATE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    add_compile_options(/utf-8)
endif()
enable_testing()
add_executable(core-tests tests.cpp)
target_link_libraries(core-tests PRIVATE vterm)
add_test(NAME core-tests COMMAND core-tests)
if(WIN32)
    add_executable(codex-monitor main.cpp)
    target_compile_definitions(codex-monitor PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
    target_link_libraries(codex-monitor PRIVATE vterm winhttp gdiplus ole32)
endif()
